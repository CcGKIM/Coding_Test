```python
import sys

# 입력 받기
R, C, T = map(int, sys.stdin.readline().split())
matrix = [list(map(int, sys.stdin.readline().split())) for _ in range(R)]

dust = {}
air_purifier = []

# 미세먼지 및 공기청정기 위치 저장
for i in range(R):
    for j in range(C):
        if matrix[i][j] > 0:
            dust[(i, j)] = matrix[i][j]
        if matrix[i][j] == -1:
            air_purifier.append((i, j))

# 공기청정기 위치 정렬
air_purifier.sort()

def spread_dust():
    """미세먼지 확산"""
    dx = [-1, 1, 0, 0]
    dy = [0, 0, -1, 1]

    updates = []

    for x in range(R):
        for y in range(C):
            if matrix[x][y] <= 0:
                continue
            tmp = []
            for k in range(4):
                nx, ny = x + dx[k], y + dy[k]
                if 0 <= nx < R and 0 <= ny < C and matrix[nx][ny] >= 0:
                    tmp.append((nx, ny))

            if not tmp:
                continue

            spread_amount = matrix[x][y] // 5
            updates.append((x, y, -spread_amount * len(tmp)))
            for nx, ny in tmp:
                updates.append((nx, ny, spread_amount))

    for x, y, delta in updates:
        matrix[x][y] += delta

def operate_purifier():
    """공기청정기 작동"""
    upper, lower = air_purifier

    def rotate(positions, clockwise=True):
        """주어진 위치 리스트를 한 칸씩 이동 (공기청정기 작동 효과)"""
        if clockwise:
            for i in range(len(positions) - 1, 0, -1):
                x1, y1 = positions[i]
                x2, y2 = positions[i - 1]
                if (x1, y1) == upper:
                    matrix[x2][y2] = 0
                else:
                    matrix[x1][y1] = matrix[x2][y2]
        else:
            for i in range(len(positions) - 1, 0, -1):
                x1, y1 = positions[i]
                x2, y2 = positions[i - 1]
                if (x1, y1) == lower:
                    matrix[x2][y2] = 0
                else:
                    matrix[x1][y1] = matrix[x2][y2]

    # 상부 공기청정기 이동 경로
    x, y = upper
    upper_path = [(x, y+1)]
    for i in range(y+1, C-1): upper_path.append((x, i+1))
    for i in range(x, 0, -1): upper_path.append((i-1, C-1))
    for i in range(C-1, 0, -1): upper_path.append((0, i-1))
    for i in range(0, x): upper_path.append((i+1, 0))

    # 하부 공기청정기 이동 경로
    x, y = lower
    lower_path = [(x, y+1)]
    for i in range(y+1, C-1): lower_path.append((x, i+1))
    for i in range(x, R-1): lower_path.append((i+1, C-1))
    for i in range(C-1, 0, -1): lower_path.append((R-1, i-1))
    for i in range(R-1, x, -1): lower_path.append((i-1, 0))

    rotate(upper_path, clockwise=True)
    rotate(lower_path, clockwise=False)

# T초 동안 시뮬레이션 실행
for _ in range(T):
    spread_dust()
    operate_purifier()

# 남아있는 미세먼지 양 계산
answer = sum(sum(row) for row in matrix if sum(row) > 0)
print(answer)
```
