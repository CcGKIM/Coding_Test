```python
from collections import deque
import sys

# 입력 받기
N = int(input())
array = list(map(int, sys.stdin.readline().split()))
queue = [i for i in range(1, N + 1)]  # 1부터 N까지의 번호를 큐에 저장

# 결과 저장 리스트
result = []
idx = 0

# 첫 번째 풍선 선택
ball = queue.pop(0)
result.append(ball)

go = array[ball - 1]
N = len(queue)
if go > 0:
    idx = (idx + go - 1) % N if N > 0 else 0
else:
    idx = (idx + go) % N if N > 0 else 0

# 풍선 터뜨리기 반복
while queue:
    ball = queue.pop(idx)
    result.append(ball)

    if not queue:
        break

    go = array[ball - 1]
    N = len(queue)
    if go > 0:
        idx = (idx + go - 1) % N if N > 0 else 0
    else:
        idx = (idx + go) % N if N > 0 else 0

# 결과 출력
print(*result)
```
