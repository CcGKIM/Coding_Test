```python
import heapq

N = int(input())
array = []

for _ in range(N):
    d, n = map(int, input().split())
    array.append((d, n))

# 마감 기한을 기준으로 정렬
array.sort()

heap = []
for d, n in array:
    heapq.heappush(heap, (n, d))

    # 마감 기한을 초과하면 가장 작은 컵라면 개수 제거
    if len(heap) > d:
        heapq.heappop(heap)

# 최종적으로 남은 컵라면 개수의 합 계산
answer = sum([x for x, _ in heap])
print(answer)
```
