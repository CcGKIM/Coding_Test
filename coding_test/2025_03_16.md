from bisect import bisect_left
import sys

def longest_decreasing_subsequence(N, array):
    """
    최장 감소 부분 수열 (LDS, Longest Decreasing Subsequence) 길이를 구하는 함수
    :param N: 배열의 크기
    :param array: 원본 배열
    :return: 최장 감소 부분 수열의 길이
    """
    # 배열을 뒤집어서 최장 증가 부분 수열 (LIS) 문제로 변환
    array = array[::-1]

    tmp = []  # LIS를 구하기 위한 리스트
    dp = []   # 각 단계에서의 LIS 길이를 저장하는 리스트

    for number in array:
        idx = bisect_left(tmp, number)

        if idx >= len(tmp):
            tmp.append(number)
        else:
            tmp[idx] = number

        if dp:
            dp.append(max(dp[-1], len(tmp)))
        else:
            dp.append(len(tmp))

    return dp[-1]

if __name__ == "__main__":
    N = int(input().strip())
    array = list(map(int, sys.stdin.readline().split()))
    result = longest_decreasing_subsequence(N, array)
    print(result)
