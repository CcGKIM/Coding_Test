```python
import sys

def find_max_score(matrix):
    """
    백트래킹을 이용하여 최적의 선수 배치를 찾아 최대 점수를 계산하는 함수.
    :param matrix: 11x11 능력치 행렬
    :return: 가능한 최대 점수
    """
    answer = -float('inf')
    used = [False] * 11

    def dfs(start, result):
        nonlocal answer
        if start == 11:
            answer = max(answer, result)
            return

        for col in range(11):
            if not used[col] and matrix[start][col] != 0:
                used[col] = True
                dfs(start + 1, result + matrix[start][col])
                used[col] = False

    dfs(0, 0)
    return answer

def main():
    """
    입력을 받아 여러 테스트 케이스에 대해 최대 점수를 계산하고 출력하는 메인 함수.
    """
    C = int(input())  # 테스트 케이스 개수
    results = []

    for _ in range(C):
        matrix = [list(map(int, sys.stdin.readline().split())) for _ in range(11)]
        results.append(find_max_score(matrix))

    for result in results:
        print(result)

if __name__ == "__main__":
    main()
```
