```python
import sys

# 입력 받기
R, C = map(int, sys.stdin.readline().split())
matrix = [list(sys.stdin.readline().strip()) for _ in range(R)]

# 이동 방향 (상, 하, 좌, 우)
dx = (-1, 1, 0, 0)
dy = (0, 0, -1, 1)

# 침수될 땅 저장 리스트
land_list = []

# 각 칸을 확인하여 주변 3칸 이상이 물이면 침수 대상으로 추가
for row in range(R):
    for col in range(C):
        if matrix[row][col] == 'X':
            cnt = 0
            for i in range(4):
                nx, ny = row + dx[i], col + dy[i]
                if not (0 <= nx < R and 0 <= ny < C) or matrix[nx][ny] == ".":
                    cnt += 1
            if cnt >= 3:
                land_list.append((row, col))

# 침수 대상 좌표를 '.'으로 변경
for row, col in land_list:
    matrix[row][col] = "."

# 최소, 최대 x, y 좌표 찾기
min_x, min_y, max_x, max_y = float('inf'), float('inf'), -float('inf'), -float('inf')
for row in range(R):
    for col in range(C):
        if matrix[row][col] == 'X':
            min_x = min(min_x, row)
            min_y = min(min_y, col)
            max_x = max(max_x, row)
            max_y = max(max_y, col)

# 축소된 지도 출력
for row in range(min_x, max_x + 1):
    print(''.join(matrix[row][min_y:max_y + 1]))
```
